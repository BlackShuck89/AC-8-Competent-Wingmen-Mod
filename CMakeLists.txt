cmake_minimum_required(VERSION 3.18)
project(AC8RoleAwareWingmen VERSION 0.2.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(AC8RoleWingmen SHARED src/ac8rw.cpp)
target_include_directories(AC8RoleWingmen PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_definitions(AC8RoleWingmen PRIVATE AC8RW_BUILD)
if(MSVC)
    target_compile_options(AC8RoleWingmen PRIVATE /W4)
else()
    target_compile_options(AC8RoleWingmen PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(ac8rw_test tests/test_native.cpp)
target_link_libraries(ac8rw_test PRIVATE AC8RoleWingmen)
enable_testing()
add_test(NAME native_planner COMMAND ac8rw_test)
